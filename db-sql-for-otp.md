# SQL For OTP Verification And Pedo Registration Storage

Run this script once in the Dantam Postgres database before wiring the OTP flow.

```sql
BEGIN;

CREATE TABLE IF NOT EXISTS public.phone_otp_requests (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  country_code varchar(4) NOT NULL DEFAULT '91',
  phone_number varchar(10) NOT NULL,
  whatsapp_number varchar(15) NOT NULL,
  purpose text NOT NULL,
  otp_hash text NOT NULL,
  attempt_count integer NOT NULL DEFAULT 0,
  max_attempts integer NOT NULL DEFAULT 5,
  expires_at timestamptz NOT NULL,
  verified_at timestamptz,
  request_ip_hash text,
  user_agent text,
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT phone_otp_requests_phone_number_check CHECK (phone_number ~ '^[0-9]{10}$'),
  CONSTRAINT phone_otp_requests_country_code_check CHECK (country_code = '91'),
  CONSTRAINT phone_otp_requests_whatsapp_number_check CHECK (whatsapp_number ~ '^91[0-9]{10}$'),
  CONSTRAINT phone_otp_requests_attempt_count_check CHECK (attempt_count >= 0),
  CONSTRAINT phone_otp_requests_max_attempts_check CHECK (max_attempts > 0)
);

CREATE INDEX IF NOT EXISTS phone_otp_requests_lookup_idx
  ON public.phone_otp_requests (country_code, phone_number, purpose, created_at DESC);

CREATE INDEX IF NOT EXISTS phone_otp_requests_rate_limit_idx
  ON public.phone_otp_requests (country_code, phone_number, purpose, created_at);

CREATE INDEX IF NOT EXISTS phone_otp_requests_expiry_idx
  ON public.phone_otp_requests (expires_at);

CREATE TABLE IF NOT EXISTS public.pedo_consultation_registrations (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  campaign text NOT NULL DEFAULT 'Pedo Dentist Free Consultation',
  preferred_date date NOT NULL,
  selected_date_label text,
  parent_name text NOT NULL,
  phone_country_code varchar(4) NOT NULL DEFAULT '91',
  phone_number varchar(10) NOT NULL,
  whatsapp_number varchar(15) NOT NULL,
  children jsonb NOT NULL,
  child_name text,
  child_age text,
  concern text,
  other_concern text,
  child_not_present boolean NOT NULL DEFAULT false,
  phone_verified boolean NOT NULL DEFAULT true,
  phone_verified_at timestamptz,
  otp_request_id bigint REFERENCES public.phone_otp_requests(id) ON DELETE SET NULL,
  n8n_status_code integer,
  n8n_response jsonb,
  source text NOT NULL DEFAULT 'dantamdentalcare.com/pedo-dentist-free-consultation',
  requester_ip inet,
  user_agent text,
  submitted_at timestamptz NOT NULL DEFAULT now(),
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pedo_consultation_registrations_phone_number_check CHECK (phone_number ~ '^[0-9]{10}$'),
  CONSTRAINT pedo_consultation_registrations_country_code_check CHECK (phone_country_code = '91'),
  CONSTRAINT pedo_consultation_registrations_whatsapp_number_check CHECK (whatsapp_number ~ '^91[0-9]{10}$'),
  CONSTRAINT pedo_consultation_registrations_children_array_check CHECK (jsonb_typeof(children) = 'array'),
  CONSTRAINT pedo_consultation_registrations_children_not_empty_check CHECK (jsonb_array_length(children) > 0)
);

CREATE INDEX IF NOT EXISTS pedo_consultation_registrations_phone_idx
  ON public.pedo_consultation_registrations (phone_number, submitted_at DESC);

CREATE INDEX IF NOT EXISTS pedo_consultation_registrations_date_idx
  ON public.pedo_consultation_registrations (preferred_date, submitted_at DESC);

CREATE INDEX IF NOT EXISTS pedo_consultation_registrations_created_at_idx
  ON public.pedo_consultation_registrations (created_at DESC);

COMMIT;
```

Expected tables after running:

- `public.phone_otp_requests`
- `public.pedo_consultation_registrations`

The app should send the OTP webhook using `whatsapp_number` in the format `91XXXXXXXXXX`.
